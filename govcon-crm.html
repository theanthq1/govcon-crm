Google Sheets sync
Backup your CRM data automatically
One-time setup — 5 minutes
1
Create a Google Sheet
Go to sheets.new and create a blank spreadsheet. Name it something like "GovCon CRM".
2
Open Apps Script
In the sheet, click Extensions → Apps Script. A code editor opens in a new tab.
3
Paste the script
Delete all existing code in the editor, then paste the code below and click Save (Ctrl+S).
function doPost(e) { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(data.sheet) || ss.insertSheet(data.sheet); if (data.action === "append") { if (sheet.getLastRow() === 0) sheet.appendRow(data.headers); sheet.appendRow(data.row); } else if (data.action === "delete") { var rows = sheet.getDataRange().getValues(); for (var i = rows.length - 1; i >= 1; i--) { if (rows[i][0] == data.id) { sheet.deleteRow(i + 1); break; } } } else if (data.action === "update") { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] == data.id) { sheet.getRange(i+1,1,1,data.row.length).setValues([data.row]); break; } } } return ContentService.createTextOutput(JSON.stringify({status:"ok"})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var result = {}; ["Contacts","Opportunities","Tasks"].forEach(function(name) { var sheet = ss.getSheetByName(name); if (sheet && sheet.getLastRow() > 1) { var d = sheet.getDataRange().getValues(); var h = d[0]; result[name] = d.slice(1).map(function(row) { var obj = {}; h.forEach(function(k,i){ obj[k] = row[i]; }); return obj; }); } else { result[name] = []; } }); return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
4
Deploy as a web app
Click Deploy → New deployment. Set type to Web app. Under "Who has access" choose Anyone. Click Deploy and authorize when prompted.
5
Paste the Web App URL below
Copy the URL that appears after deploying (starts with https://script.google.com/macros/s/...) and paste it here.
